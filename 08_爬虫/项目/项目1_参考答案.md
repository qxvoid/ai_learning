## 完整脚本 `project1_crawler.py`

```python
# -*- coding: utf-8 -*-
"""项目1 参考答案：books.toscrape.com 全站采集 -> 清洗 -> SQLite -> SQL 分析"""
import re
import time
import sqlite3
from datetime import datetime
from urllib.parse import urljoin

import requests
from bs4 import BeautifulSoup

BASE = "https://books.toscrape.com/"
START = "https://books.toscrape.com/catalogue/page-1.html"
DB_PATH = "books.db"
DELAY = 1.0  # 礼貌延迟（秒）

STAR_MAP = {"One": 1, "Two": 2, "Three": 3, "Four": 4, "Five": 5}

session = requests.Session()
session.headers.update({
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                  "AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0 Safari/537.36"
})


def clean_price(text: str) -> float:
    return float(re.sub(r"[^\d.]", "", text))


def parse_list_page(html: str, base_url: str):
    """解析一页列表，返回 (记录列表, 下一页URL或None)。"""
    soup = BeautifulSoup(html, "lxml")
    records = []
    for pod in soup.select("article.product_pod"):
        title = pod.h3.a.get("title").strip()
        price = clean_price(pod.select_one("p.price_color").get_text())
        star_cls = pod.select_one("p.star-rating").get("class")
        star = STAR_MAP.get(star_cls[-1], 0) if star_cls else 0
        detail_url = urljoin(base_url, pod.h3.a.get("href"))
        records.append({
            "title": title, "price": price, "star": star,
            "detail_url": detail_url,
            "crawled_at": datetime.now().isoformat(timespec="seconds"),
        })
    next_a = soup.select_one("li.next a")
    next_url = urljoin(base_url, next_a.get("href")) if next_a else None
    return records, next_url


def fetch_upc(detail_url: str) -> str:
    """访问详情页抓 UPC（加分项）。"""
    time.sleep(DELAY)
    r = session.get(detail_url, timeout=10)
    soup = BeautifulSoup(r.text, "lxml")
    return soup.select_one("table.table tr td").get_text(strip=True)


def init_db():
    conn = sqlite3.connect(DB_PATH)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS books (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            title TEXT UNIQUE,
            price REAL,
            star INTEGER,
            upc TEXT,
            detail_url TEXT,
            crawled_at TEXT
        )
    """)
    conn.commit()
    return conn


def save_records(conn, records):
    cur = conn.executemany(
        "INSERT OR IGNORE INTO books (title, price, star, upc, detail_url, crawled_at) "
        "VALUES (:title, :price, :star, :upc, :detail_url, :crawled_at)",
        records,
    )
    conn.commit()
    return cur.rowcount


def main():
    conn = init_db()
    url = START
    page = 0
    while url:
        page += 1
        print(f"第 {page} 页: {url}")
        r = session.get(url, timeout=10)
        records, url = parse_list_page(r.text, url)
        for rec in records:
            rec["upc"] = fetch_upc(rec["detail_url"])   # 加分项：详情页
        save_records(conn, records)
        time.sleep(DELAY)

    # ---- SQL 分析 ----
    cur = conn.cursor()
    print("\n=== 分析 ===")
    cur.execute("SELECT COUNT(*) FROM books"); print("总书数:", cur.fetchone()[0])
    cur.execute("SELECT title, price FROM books ORDER BY price DESC LIMIT 10")
    print("最贵 10 本:", cur.fetchall())
    cur.execute("SELECT AVG(price) FROM books"); print("平均价: %.2f" % cur.fetchone()[0])
    cur.execute("SELECT star, COUNT(*) FROM books GROUP BY star ORDER BY star DESC")
    print("按星级:", cur.fetchall())
    cur.execute("SELECT COUNT(*) FROM books WHERE price < 30 AND star >= 4")
    print("低价高分书(price<30 & star>=4):", cur.fetchone()[0])
    conn.close()


if __name__ == "__main__":
    main()
```

---

## 关键设计说明（对照任务书看）

- **翻页**：`parse_list_page` 同时返回记录和下一页 URL，主 while 循环靠 `next_url` 自驱，没有写死页数。
- **去重/增量**：`title` 设 UNIQUE + `INSERT OR IGNORE`，重复运行不会重复写。
- **参数化**：用命名占位符 `:title`，值走 dict，绝不 f-string 拼 SQL。
- **限速**：每个列表页、每个详情页之间都 `time.sleep(DELAY)`。
- **清洗**：`clean_price` 去货币符号，星级映射成整数。
